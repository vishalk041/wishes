    <script>

        // Store user's name

        let userName = "";


        // Function to switch between stages

        function switchStage(hideId, showId) {

            const hideElement = document.getElementById(hideId);

            const showElement = document.getElementById(showId);

            

            hideElement.classList.remove('active');

            

            setTimeout(() => {

                hideElement.style.display = 'none';

                showElement.style.display = 'flex';

                // Small delay to allow display:flex to apply before adding opacity

                setTimeout(() => {

                    showElement.classList.add('active');

                }, 50);

            }, 500); // Wait for fade out

        }


        // Action for Stage 1 -> Stage 2

        function goToStage2() {

            const input = document.getElementById('nameInput');

            const errorMsg = document.getElementById('nameError');

            const nameVal = input.value.trim().toLowerCase();

            

            if(nameVal !== "anjali") {

                input.classList.add('border-red-400', 'animate-pulse');

                errorMsg.classList.remove('hidden');

                setTimeout(() => input.classList.remove('border-red-400', 'animate-pulse'), 1000);

                return;

            }

            

            errorMsg.classList.add('hidden');

            userName = "Anjali";

            

            // Update greeting

            document.getElementById('greetingTitle').innerText = `Hey ${userName}! ✨`;

            

            switchStage('stage1', 'stage2');

        }


        // Action for Stage 2 -> Stage 3

        function goToStage3() {

            switchStage('stage2', 'stage3');

        }


        // Action for Stage 3 -> Stage 4

        function goToStage4() {

            switchStage('stage3', 'stage4');

            generateFireworks(); // Add a little celebration

        }


        const btnNo = document.getElementById('btnNo');


        function moveButton() {

            // Change position to fixed so it moves relative to the screen

            btnNo.style.position = 'fixed';

            

            // Get window dimensions

            const windowWidth = window.innerWidth;

            const windowHeight = window.innerHeight;

            

            // Get button dimensions

            const btnWidth = btnNo.offsetWidth;

            const btnHeight = btnNo.offsetHeight;

            

            // Calculate random positions keeping button strictly within view

            // Added 20px padding from edges to guarantee it won't trigger scrollbars or get cut off

            const maxX = windowWidth - btnWidth - 40; 

            const maxY = windowHeight - btnHeight - 40;

            

            const randomX = Math.max(20, Math.floor(Math.random() * maxX));

            const randomY = Math.max(20, Math.floor(Math.random() * maxY));

            

            // Apply new coordinates

            btnNo.style.left = randomX + 'px';

            btnNo.style.top = randomY + 'px';

        }


        // Trigger evasion on mouse hover (Desktop) and touch (Mobile)

        btnNo.addEventListener('mouseover', moveButton);

        btnNo.addEventListener('touchstart', function(e) {

            e.preventDefault(); // Prevents the actual click/tap

            moveButton();

        });


        function createParticles() {

            const container = document.getElementById('particles-container');

            const particleCount = 15;

            const symbols = ['✨', '💖', '🌸', '🤍', '💫'];


            for (let i = 0; i < particleCount; i++) {

                let particle = document.createElement('div');

                particle.className = 'particle';

                particle.innerText = symbols[Math.floor(Math.random() * symbols.length)];

                

                // Randomize position, size, and animation duration

                particle.style.left = Math.random() * 100 + 'vw';

                particle.style.fontSize = (Math.random() * 15 + 10) + 'px';

                particle.style.animationDuration = (Math.random() * 5 + 5) + 's';

                particle.style.animationDelay = (Math.random() * 5) + 's';

                

                container.appendChild(particle);

            }

        }


        function createLoveBubbles() {

            const container = document.getElementById('particles-container');

            const bubbleCount = 25;


            for (let i = 0; i < bubbleCount; i++) {

                let bubble = document.createElement('div');

                bubble.className = 'love-bubble';

                

                // Randomize size, position, and animation timing

                let size = Math.random() * 45 + 10; // Sizes between 10px and 55px

                bubble.style.width = size + 'px';

                bubble.style.height = size + 'px';

                bubble.style.left = Math.random() * 100 + 'vw';

                

                // Set durations for floating up and swaying side-to-side

                let floatDuration = Math.random() * 8 + 6;

                let swayDuration = Math.random() * 2 + 2;

                bubble.style.animationDuration = floatDuration + 's, ' + swayDuration + 's';

                

                // Set delays so they don't all appear at once

                let floatDelay = Math.random() * 10;

                let swayDelay = Math.random() * 2;

                bubble.style.animationDelay = floatDelay + 's, ' + swayDelay + 's';

                

                container.appendChild(bubble);

            }

        }


        function generateFireworks() {

            const container = document.getElementById('particles-container');

            for (let i = 0; i < 20; i++) {

                let confetti = document.createElement('div');

                confetti.className = 'particle';

                confetti.innerText = '🎉';

                confetti.style.left = Math.random() * 100 + 'vw';

                confetti.style.animationDuration = (Math.random() * 3 + 2) + 's';

                container.appendChild(confetti);

            }

        }


        // Initialize particles on load

        window.onload = () => {

            createParticles();

            createLoveBubbles(); // Add the bubbles to the initial setup

            

            // Handle Enter key for input

            document.getElementById('nameInput').addEventListener('keypress', function (e) {

                if (e.key === 'Enter') {

                    goToStage2();

                }

            });

        };
